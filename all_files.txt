async function usernameToId(handle) {
	const h = handle.replace(/^@/, "").trim();

	// Primary: modern endpoint
	try {
		const r = await fetch("https://users.roblox.com/v1/usernames/users", {
			method: "POST",
			headers: { "content-type": "application/json" },
			body: JSON.stringify({ usernames: [h], excludeBannedUsers: false })
		});
		if (r.ok) {
			const j = await r.json();
			const id = j?.data?.[0]?.id;
			if (id) return id;
		}
	} catch (_) { }

	// Fallback: legacy endpoint
	const r2 = await fetch(
		"https://api.roblox.com/users/get-by-username?username=" + encodeURIComponent(h)
	);
	if (!r2.ok) return null;
	const j2 = await r2.json();
	return j2?.Id || j2?.id || null;
}

// ---------------------------------------------- CALCULATIONS

const toKebab = key => key.replace(/([A-Z])/g, "-$1").toLowerCase()

// ---------------------------------------------- ACTIONS - Events / Listeners / Observers / Intervals / OnLoad

const windowOnLoadFuncs = []

const windowEvents = {}
const recordWindowEvent = (name, func) => {
	if (!windowEvents[name]) windowEvents[name] = []
	windowEvents[name].push(func)
}

const documentEvents = {}
const recordDocumentEvent = (name, func) => {
	if (!documentEvents[name]) documentEvents[name] = []
	documentEvents[name].push(func)
}

const prepare = () => {
	// Add window and document event listeners:
	windowOnLoadFuncs.push(() => {

		// Window event listeners:
		for (const eventName in windowEvents) {
			window.addEventListener(eventName, e => {
				const funcs = windowEvents[eventName]
				funcs.forEach(func => func(e))
			})
		}

		// Document event listeners:
		for (const eventName in documentEvents) {
			document.addEventListener(eventName, e => {
				const funcs = documentEvents[eventName]
				funcs.forEach(func => func(e))
			})
		}
	})

	// Add onload event:
	window.onload = () => Promise.all(windowOnLoadFuncs.map(fn => fn()))
}

// ---------------------------------------------- HTML - Base Functions

const dynamicStyle = document.createElement("style")
document.head.appendChild(dynamicStyle)
const cssRules = new Set()
const addCssRule = rule => {
	if (cssRules.has(rule)) return;
	try {
		dynamicStyle.sheet.insertRule(rule, dynamicStyle.sheet.cssRules.length);
		cssRules.add(rule);
	} catch (err) {
		console.error("insertRule failed for:", rule, err);
	}
};

const getImgClass = (imgSrc) => "img_" + imgSrc.split("/").pop().replace(/[^\w]+/g, "_");

function preloadImages(sources) {
	return Promise.all(sources.map(src => new Promise(res => {
		// make class name from filename
		const name = getImgClass(src)
		addCssRule(`.${name} {
			background-image: url("${src}");
			background-repeat: no-repeat;
			background-size: cover;
			background-position: center;
		}`);
		// warm cache
		const img = new Image();
		img.decoding = "async";
		img.onload = img.onerror = res;
		img.src = src;
	})));
}

function makeHtml({ tag, style = {}, props = {}, on = {}, children = [] }) {
	const el = document.createElement(tag)

	// css
	const baseStyle = {}
	const pseudoStyles = {}
	for (const [k, v] of Object.entries(style)) {
		if (k.startsWith(":")) pseudoStyles[k.slice(1)] = v
		else baseStyle[k] = v
	}

	if (Object.keys(pseudoStyles).length === 0) {
		// no pseudos → everything inline
		for (const [k, v] of Object.entries(baseStyle)) {
			el.style[k] = v
		}
	} else {
		// pseudos present → class + rules
		const hashInput = JSON.stringify({ base: baseStyle, pseudos: pseudoStyles })
		const hash = btoa(hashInput).replace(/[^a-z0-9]/gi, "")
		const className = `auto-style-${hash}`
		el.classList.add(className)

		// base rule
		const baseCss = Object.entries(baseStyle)
			.map(([k, v]) => `${toKebab(k)}:${v};`)
			.join(" ")
		addCssRule(`.${className} { ${baseCss} }`)

		// pseudo rules
		for (const [pseudo, psStyles] of Object.entries(pseudoStyles)) {
			const cssBody = Object.entries(psStyles)
				.map(([k, v]) => `${toKebab(k)}:${v} !important;`)
				.join(" ")
			addCssRule(`.${className}:${pseudo} { ${cssBody} }`)
		}
	}

	// props
	for (const [k, v] of Object.entries(props)) {
		if (k === "class" || k === "className") {
			const cls = Array.isArray(v) ? v : String(v).split(/\s+/).filter(Boolean);
			if (cls.length) el.classList.add(...cls);
			continue;
		}
		if (k in el) el[k] = v;
		else el.setAttribute(k, v);
	}

	// events
	for (const [evt, fn] of Object.entries(on)) {
		el.addEventListener(evt, fn)
	}

	// children
	for (const child of children) {
		el.appendChild(
			typeof child === "string"
				? document.createTextNode(child)
				: makeHtml(child)
		)
	}

	return el
}

			left: "50%",

			fontSize: "0.9rem",
			zIndex: 9999,
			opacity: "0",

		},
		children: [msg]

	// fade out and remove
	setTimeout(() => {

// --- singleton picker
let pickerEl = null
let pickerOpen = false

function hidePicker() {
	if (!pickerEl) return
	pickerEl.style.display = "none"
	pickerOpen = false
}

function showPicker({ items, onPick, posX, posY }) {
	console.log({ pickerEl, pickerOpen })

	// Create picker if it doesnt exist:
	if (!pickerEl) {
		pickerEl = makeHtml({
			tag: "div",
			style: {
				position: "fixed",
				left: "0px", top: "0px",
				background: "rgba(0,0,0,0.9)",
				color: "white",
				padding: "0.5rem",
				borderRadius: "0.5rem",
				zIndex: 9999,
				display: "none",
				gridTemplateColumns: "repeat(5, 2rem)",
				gap: "0.5rem",
			},
			props: { id: "picker" }
		})
		document.body.appendChild(pickerEl)

		// one-time outside/Esc handlers
		document.addEventListener("mousedown", (e) => {
			if (!pickerOpen) return
			if (!pickerEl.contains(e.target)) hidePicker()
		})
		document.addEventListener("keydown", (e) => {
			if (!pickerOpen) return
			if (e.key === "Escape") hidePicker()
		})
	}

	// Setup picker position and options:
	pickerEl.style.left = posX + "px"
	pickerEl.style.top = posY + "px"
	pickerEl.replaceChildren(...items.map(item => (
		makeHtml({
			tag: "button",
			children: [
				{ tag: "div", props: { class: getImgClass(item.imgSrc) }, style: { width: "2rem", height: "2rem" } }
			],
			on: {
				click: () => {
					onPick(item)
					hidePicker()
				}
			}
		})
	)))

	// Show picker:
	pickerEl.style.display = "grid"
	pickerOpen = true

	console.log({ pickerEl, pickerOpen })
}
